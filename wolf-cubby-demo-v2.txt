// ─── HELPERS ─────────────────────────────────────────────────────────────────

function siteHealth(site){
  let h='ok';
  for(const s of site.skus){
    if(s.q<=0||(s.q<s.min))h='crit';
    else if(h!=='crit'&&s.q<=s.min+2)h='low';
  }
  return h;
}

function avgUsage(sku){
  return sku.usage.reduce((a,b)=>a+b,0)/sku.usage.length;
}

function daysToMin(sku){
  const rate=avgUsage(sku);
  if(rate<=0)return 999;
  return Math.round((sku.q-sku.min)/rate*7);
}

function predClass(days){
  if(days<=0)return'crit';
  if(days<=14)return'warn';
  return'ok';
}

function predLabel(days){
  if(days<=0)return'Below min now';
  if(days===999)return'Stable';
  return'Hits min in ~'+days+'d';
}

function healthLabel(h){return h==='ok'?'OK':h==='low'?'Low':'Critical'}

let toastTimer;
function toast(m){
  const t=document.getElementById('toast');
  t.textContent=m;t.style.opacity=1;
  clearTimeout(toastTimer);
  toastTimer=setTimeout(()=>t.style.opacity=0,2200);
}

  // summary row at top

  const lowSites=SITES.filter(s=>siteHealth(s)==='low').length;
  const totalSkus=SITES.reduce((a,s)=>a+s.skus.length,0);

  let html='<div class="sg" style="margin-bottom:1px">'+
    '<div class="st"><div class="sv" style="color:var(--b)">'+totalSites+'</div><div class="sl">Client sites</div></div>'+
    '<div class="st"><div class="sv" style="color:var(--b)">'+totalSkus+'</div><div class="sl">SKUs on consign</div></div>'+
    '<div class="st"><div class="sv" style="color:'+(critSites>0?'var(--r)':'var(--g)')+'">'+critSites+'</div><div class="sl">Critical</div></div>'+
    '<div class="st"><div class="sv" style="color:'+(lowSites>0?'var(--a)':'var(--g)')+'">'+lowSites+'</div><div class="sl">Low stock</div></div>'+
  '</div>';

  html+='<div class="sc"><div class="scl">All client sites</div></div>';
  html+='<div class="sites-grid">';

  SITES.forEach(site=>{
    const h=siteHealth(site);
    const hLabel=healthLabel(h);
    const autoIcon=site.autoOrder?'<i class="ti ti-refresh site-auto on" title="Auto-order on"></i>':'<i class="ti ti-refresh site-auto off" title="Auto-order off"></i>';

function renderSdStock(site){
  let html='';
  site.skus.forEach(sku=>{
    const pct=Math.min(100,Math.round((sku.q/sku.max)*100));
    const minPct=Math.round((sku.min/sku.max)*100);
    const fillCol=sku.q<sku.min?'var(--r)':sku.q<=sku.min+2?'var(--a)':'var(--g)';
    const days=daysToMin(sku);
    const pc=predClass(days);
    const rate=avgUsage(sku).toFixed(1);
    html+='<div class="ssite-row">'+
      '<div class="ssite-top">'+
        '<div class="ssite-name">'+sku.nm+'</div>'+
        '<div class="ssite-qty" style="color:'+fillCol+'">'+sku.q+'</div>'+
      '</div>'+
      // min/max bar
      '<div class="mmbar-wrap">'+
        '<div class="mmbar-fill" style="width:'+pct+'%;background:'+fillCol+'"></div>'+
        '<div class="mmbar-min" style="left:'+minPct+'%"></div>'+
      '</div>'+
      '<div class="mmbar-meta">'+
        '<span>'+sku.q+' / max '+sku.max+'</span>'+
        '<span style="color:var(--a)">min '+sku.min+'</span>'+
      '</div>'+
      '<div style="display:flex;align-items:center;justify-content:space-between;margin-top:5px">'+
        '<span class="pred-badge '+pc+'"><i class="ti ti-trending-down" style="font-size:10px"></i>'+predLabel(days)+'</span>'+
        '<span style="font-family:var(--m);font-size:9px;color:var(--tx3)">'+rate+' units/wk avg</span>'+
      '</div>'+
    '</div>';
  });
  document.getElementById('sdStock').innerHTML=html+'<div class="spacer"></div>';
}

function renderSdReplen(site){

  const maxRate=Math.max(...site.skus.map(s=>avgUsage(s)));

    const rate=avgUsage(sku);
    const h=Math.max(4,Math.round((rate/Math.max(maxRate,1))*56));
    chartHtml+='<div class="usage-bar-wrap">'+
      '<div class="usage-bar" style="height:'+h+'px"></div>'+
      '<div class="usage-lbl">'+sku.p.split('-')[0]+'</div>'+
    '</div>';
  });
  chartHtml+='</div>';
  document.getElementById('sdUsageChart').innerHTML=chartHtml;

  // replen queue
  let qHtml='';
  site.skus.forEach(sku=>{
    const needReplen=sku.q<sku.min;
    const replenQty=sku.max-sku.q;
    qHtml+='<div class="replen-item">'+
      '<div class="replen-info">'+
        '<div class="replen-name">'+sku.nm+'</div>'+
        '<div class="replen-meta">'+sku.q+' on site · max '+sku.max+' · order '+replenQty+' units</div>'+
      '</div>'+
      '<button class="ord-btn '+(needReplen?'need':'done')+'" onclick="raiseOrder(\''+site.id+'\',\''+sku.p+'\')">'+
        (needReplen?'Order now':'OK')+
      '</button>'+
    '</div>';
  });
  document.getElementById('sdReplenQueue').innerHTML=qHtml;
}

function renderSdSettings(site){
  let html='<div class="toggle-row"><div><div class="toggle-label">Site auto-order</div><div class="toggle-sub">Applies to all SKUs unless overridden</div></div><label class="tog"><input type="checkbox" '+(site.autoOrder?'checked':'')+' onchange="toggleSiteAuto(\''+site.id+'\',this.checked)"><div class="tog-sl"></div></label></div>';

  html+='<div class="sc" style="margin-top:4px"><div class="scl">Per-SKU auto-order</div></div>';
  site.skus.forEach(sku=>{
    html+='<div class="toggle-row"><div><div class="toggle-label">'+sku.nm+'</div><div class="toggle-sub">Min '+sku.min+' · Max '+sku.max+' · '+avgUsage(sku).toFixed(1)+' units/wk</div></div><label class="tog"><input type="checkbox" '+(sku.autoOrder?'checked':'')+' onchange="toggleSkuAuto(\''+site.id+'\',\''+sku.p+'\',this.checked)"><div class="tog-sl"></div></label></div>';
  });

  document.getElementById('sdSettingsContent').innerHTML=html;
}

function toggleSiteAuto(siteId,val){
  const site=SITES.find(s=>s.id===siteId);if(!site)return;
  site.autoOrder=val;
  toast('Site auto-order '+(val?'enabled':'disabled'));
}
function toggleSkuAuto(siteId,skuP,val){
  const site=SITES.find(s=>s.id===siteId);if(!site)return;
  const sku=site.skus.find(s=>s.p===skuP);if(!sku)return;
  sku.autoOrder=val;
  toast(skuP+' auto-order '+(val?'on':'off'));
}
function raiseOrder(siteId,skuP){
  const site=SITES.find(s=>s.id===siteId);
  const sku=site?site.skus.find(s=>s.p===skuP):null;
  if(!sku)return;
  const qty=sku.max-sku.q;
  toast('Order raised: '+skuP+' × '+qty+' → '+site.name);
}
